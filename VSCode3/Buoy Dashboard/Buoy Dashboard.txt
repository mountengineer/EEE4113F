// ═══════════════════════════════════════════════════════════════
//  MAP INITIALISATION

let map;

function initMap() {
  map = L.map('map', {
    center: [-62.5, -58.8],
    zoom: 7,
    zoomControl: true
  });

  // Dark ocean tile layer
  L.tileLayer('https://{s}.basemaps.cartocdn.com/dark_all/{z}/{x}/{y}{r}.png', {
    attribution: '© OpenStreetMap © CARTO',
    maxZoom: 18
  }).addTo(map);
}

function updateMarkers() {
  Object.values(buoys).forEach(b => {
    const markerCss = MARKER_CSS[b.mode] || 'marker-sleep';
    const html = `<div class="buoy-marker ${markerCss}">${b.id}</div>`;
    const icon = L.divIcon({ html, className:'', iconSize:[36,36], iconAnchor:[18,18] });

    const ageMins = Math.round((Date.now() - b.lastSeen) / 60000);
    const popupHtml = `
      <div class="popup-inner">
        <div class="popup-title">Buoy ${b.id}</div>
        <div class="popup-row"><span class="k">Mode</span><span class="v">${MODE_NAMES[b.mode]}</span></div>
        <div class="popup-row"><span class="k">Battery</span><span class="v">${b.battery.toFixed(1)}%</span></div>
        <div class="popup-row"><span class="k">GPS Fix</span><span class="v">${b.gpsFix ? 'YES' : 'NO'}</span></div>
        <div class="popup-row"><span class="k">Hs</span><span class="v">${b.Hs ? b.Hs.toFixed(2)+' m' : '—'}</span></div>
        <div class="popup-row"><span class="k">Last seen</span><span class="v">${ageMins}m ago</span></div>
        <div class="popup-row"><span class="k">RSSI</span><span class="v">${b.loraRSSI.toFixed(1)} dBm</span></div>
      </div>`;

    if (markers[b.id]) {
      markers[b.id].setLatLng([b.lat, b.lon]);
      markers[b.id].setIcon(icon);
      markers[b.id].setPopupContent(popupHtml);
    } else {
      const m = L.marker([b.lat, b.lon], { icon })
        .addTo(map)
        .bindPopup(popupHtml);
      m.on('click', () => selectBuoy(b.id));
      markers[b.id] = m;
    }
  });
}

// ═══════════════════════════════════════════════════════════════
//  FLEET CARDS
// ═══════════════════════════════════════════════════════════════
function renderFleetCards() {
  const container = document.getElementById('buoyCardContainer');
  container.innerHTML = '';
  const sorted = Object.values(buoys).sort((a,b) => a.id.localeCompare(b.id));
  sorted.forEach(b => {
    const ageSec = Math.round((Date.now() - b.lastSeen) / 1000);
    const ageStr = ageSec < 60 ? ageSec+'s' : Math.round(ageSec/60)+'m';
    const battClass = b.battery > 50 ? 'ok' : b.battery > 20 ? 'warn' : 'val';
    const gpsClass = b.gpsFix ? 'ok' : 'warn';
    const rssiClass = b.loraRSSI > -120 ? 'ok' : b.loraRSSI > -135 ? 'warn' : 'dim';
    const card = document.createElement('div');
    card.className = 'buoy-card' + (selectedBuoy === b.id ? ' selected' : '');
    card.id = 'card-'+b.id;
    card.onclick = () => selectBuoy(b.id);
    card.innerHTML = `
      <div class="bc-header">
        <span class="bc-id">BUOY ${b.id}</span>
        <span class="mode-badge ${MODE_CSS[b.mode]}">${MODE_NAMES[b.mode]}</span>
      </div>
      <div class="bc-grid">
        <div class="bc-stat">
          <div class="lbl">Battery</div>
          <div class="val ${battClass}">${b.battery.toFixed(1)}%</div>
        </div>
        <div class="bc-stat">
          <div class="lbl">GPS Fix</div>
          <div class="val ${gpsClass}">${b.gpsFix ? 'YES' : 'NO FIX'}</div>
        </div>
        <div class="bc-stat">
          <div class="lbl">Hs</div>
          <div class="val">${b.Hs ? b.Hs.toFixed(2)+' m' : '—'}</div>
        </div>
        <div class="bc-stat">
          <div class="lbl">RSSI</div>
          <div class="val ${rssiClass}">${b.loraRSSI.toFixed(1)} dBm</div>
        </div>
      </div>
      <div class="bc-footer">
        <span>(${b.lat.toFixed(3)}, ${b.lon.toFixed(3)})</span>
        <span>Last: ${ageStr} ago</span>
      </div>`;
    container.appendChild(card);
  });
}

function renderDetail(id) {
  const b = buoys[id];
  if (!b) return;
  const ageMins = ((Date.now() - b.lastSeen) / 60000).toFixed(1);
  const battClass = b.battery > 50 ? 'ok' : b.battery > 20 ? 'warn' : 'dim';
  document.getElementById('detailContent').innerHTML = `
    <div class="detail-section">
      <div class="detail-title">Buoy ${b.id} — ${MODE_NAMES[b.mode]}</div>
      <div class="detail-grid">
        <div class="dstat"><div class="lbl">Latitude</div><div class="val">${b.lat.toFixed(5)}°</div></div>
        <div class="dstat"><div class="lbl">Longitude</div><div class="val">${b.lon.toFixed(5)}°</div></div>
        <div class="dstat"><div class="lbl">Battery</div><div class="val ${battClass}">${b.battery.toFixed(1)}%</div></div>
        <div class="dstat"><div class="lbl">GPS Fix</div><div class="val ${b.gpsFix?'ok':'warn'}">${b.gpsFix?'Acquired':'No Fix'}</div></div>
        <div class="dstat"><div class="lbl">LoRa RSSI</div><div class="val">${b.loraRSSI.toFixed(1)} dBm</div></div>
        <div class="dstat"><div class="lbl">LoRa SNR</div><div class="val">${b.loraSNR.toFixed(1)} dB</div></div>
        <div class="dstat full"><div class="lbl">Last received</div><div class="val">${ageMins} minutes ago</div></div>
      </div>
    </div>
    <div class="detail-section">
      <div class="detail-title">Wave Data</div>
      <div class="detail-grid">
        <div class="dstat full">
          <div class="lbl">Significant Wave Height (Hs)</div>
          <div class="val" style="font-size:22px;color:var(--signal)">${b.Hs ? b.Hs.toFixed(2)+' m' : '— (calibrating)'}</div>
        </div>
      </div>
      ${b.psd ? `
      <div class="psd-container">
        <div class="lbl" style="font-size:9px;margin-bottom:4px;">POWER SPECTRAL DENSITY (relative)</div>
        <canvas id="psdCanvas" width="280" height="80"></canvas>
      </div>` : '<p style="font-size:11px;color:var(--dim);margin-top:8px;">PSD not yet available</p>'}
    </div>`;

  if (b.psd) drawPSD(b.psd);
}

function drawPSD(data) {
  const canvas = document.getElementById('psdCanvas');
  if (!canvas) return;
  const ctx = canvas.getContext('2d');
  const W = canvas.width, H = canvas.height;
  ctx.clearRect(0,0,W,H);

  const max = Math.max(...data);
  const barW = W / data.length;

  // Gradient fill
  const grad = ctx.createLinearGradient(0,0,0,H);
  grad.addColorStop(0,'rgba(0,200,255,0.8)');
  grad.addColorStop(1,'rgba(0,200,255,0.1)');

  ctx.fillStyle = grad;
  data.forEach((v,i) => {
    const h = (v/max) * (H-4);
    ctx.fillRect(i*barW+1, H-h, barW-2, h);
  });

  // Axis line
  ctx.strokeStyle = 'rgba(255,255,255,0.1)';
  ctx.beginPath(); ctx.moveTo(0,H-1); ctx.lineTo(W,H-1); ctx.stroke();
}

let bleDevice = null, bleTxChar = null, bleConnected = false;
const HC08_SVC  = '0000ffe0-0000-1000-8000-00805f9b34fb';
const HC08_CHAR = '0000ffe1-0000-1000-8000-00805f9b34fb';

function openBleModal()  { document.getElementById('bleModal').classList.add('show'); }
function closeBleModal() { document.getElementById('bleModal').classList.remove('show'); }

async function bleConnect() {
  if (!navigator.bluetooth) {
    bleLog('Web Bluetooth not available. Use Chrome on Android or desktop.','err');
    document.getElementById('bleModalStatus').textContent = 'Web Bluetooth not supported in this browser';
    return;
  }
  bleLog('Scanning for HC-08...','sys');
  try {
    bleDevice = await navigator.bluetooth.requestDevice({
      filters:[{name:'HC-08'},{namePrefix:'HC-0'},{services:[HC08_SVC]}],
      optionalServices:[HC08_SVC]
    });
    bleDevice.addEventListener('gattserverdisconnected', () => {
      bleLog('Disconnected','err');
      document.getElementById('bleModalStatus').textContent = 'Disconnected';
      bleConnected = false;
    });
    const server = await bleDevice.gatt.connect();
    const svc    = await server.getPrimaryService(HC08_SVC);
    bleTxChar    = await svc.getCharacteristic(HC08_CHAR);
    await bleTxChar.startNotifications();
    bleTxChar.addEventListener('characteristicvaluechanged', e => {
      const txt = new TextDecoder().decode(new Uint8Array(e.target.value.buffer));
      bleLog('← '+txt.trim(),'rx');
      try {
        const d = JSON.parse(txt);
        if (d.nodeID) {
          document.getElementById('bleModalStatus').textContent =
            'Buoy '+d.nodeID+' | Mode:'+d.mode+' | Batt:'+d.battery.toFixed(1)+'%';
          // Inject live BLE data into dashboard state
          const id = String(d.nodeID).padStart(2,'0');
          if (buoys[id]) {
            buoys[id].battery = d.battery;
            buoys[id].mode = String(d.mode).padStart(2,'0');
            buoys[id].gpsFix = d.gpsFix;
            buoys[id].loraRSSI = d.loraRSSI || buoys[id].loraRSSI;
            buoys[id].lastSeen = Date.now();
            renderFleetCards(); updateMarkers(); updateTopBar();
          }
        }
      } catch(e) {}
    });
    bleConnected = true;
    document.getElementById('bleModalStatus').textContent = 'Connected to '+bleDevice.name;
    bleLog('Connected to '+bleDevice.name,'rx');
    // Request status automatically
    await bleTxChar.writeValue(new TextEncoder().encode('STATUS\n'));
    bleLog('→ STATUS','tx');
  } catch(err) {
    bleLog('Error: '+err.message,'err');
  }
}

function bleLog(msg, cls) {
  const log = document.getElementById('bleLog');
  const line = document.createElement('div');
  line.className = 'log-'+(cls||'sys');
  line.textContent = msg;
  log.appendChild(line);
  log.scrollTop = log.scrollHeight;
}

}
